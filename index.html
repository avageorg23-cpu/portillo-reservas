async function onDayClick(isoDate){ const room = document.getElementById("bungalow").value; // USAMOS LA CACHÉ LOCAL AL INSTANTE SIN LLAMADAS DE RED (Cero trabas) const occupied = occupiedCache.get(room) || new Set(); if (!selection.checkin || selection.checkout){ selection.checkin = isoDate; selection.checkout = null; clearCalendarHighlights(); document.querySelector(`td[data-fecha="${isoDate}"]`)?.classList.add("checkin"); updateSummary(); return; } if (isoDate <= selection.checkin){ selection.checkin = isoDate; selection.checkout = null; clearCalendarHighlights(); document.querySelector(`td[data-fecha="${isoDate}"]`)?.classList.add("checkin"); updateSummary(); return; } if (enumerateNights(selection.checkin, isoDate).some(n => occupied.has(n))){ alert("Hay una noche ocupada en el rango seleccionado."); return; } selection.checkout = isoDate; selection.nights = diffNights(selection.checkin, selection.checkout); clearCalendarHighlights(); highlightRange(selection.checkin, selection.checkout); document.querySelector(`td[data-fecha="${selection.checkin}"]`)?.classList.add("checkin"); document.querySelector(`td[data-fecha="${selection.checkout}"]`)?.classList.add("checkout"); updateSummary(); } // Y actualizamos el evento del selector de apartamentos para que precargue al instante la caché sin bloquear document.addEventListener("DOMContentLoaded", () => { const bungalowSelect = document.getElementById("bungalow"); bungalowSelect.addEventListener("change", () => { // Limpiamos selección actual al cambiar de apart selection = { checkin: null, checkout: null, nights: 0 }; document.getElementById("payment-section").style.display = "none"; loadAvailabilityData(); // Precarga silenciosa en segundo plano del nuevo apart seleccionado loadOccupiedForRoom(bungalowSelect.value, true); }); document.getElementById("moneda").addEventListener("change", updatePriceLine); loadAvailabilityData(); });
